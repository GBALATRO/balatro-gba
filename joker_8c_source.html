168ItemDescSubtypeInfo joker_object_get_rarity_info(Item* joker_object)
169{
170 ItemDescSubtypeInfo subtype_info = ITEM_SUBTYPE_INFO_DEFAULT;
171
172 GBAL_RETURN_IF_NULL(joker_object, subtype_info);
173 GBAL_RETURN_IF_ASSERT_FAILS(joker_object->type == ITEM_TYPE_JOKER, subtype_info);
174
175 Joker* joker = ((JokerObject*)joker_object)->joker;
176 GBAL_RETURN_IF_NULL(joker, subtype_info);
177
178 const JokerInfo* info = get_joker_registry_entry(joker->id);
179 GBAL_RETURN_IF_NULL(info, subtype_info);
180
181 u8 rarity = joker->rarity;
182 if (rarity >= MAX_RARITIES)
183 {
184 MGBA_FUNC_ERROR(
185 "Invalid Joker rarity value %d, should not exceed %d",
186 rarity,
187 MAX_RARITIES
188 );
189 return subtype_info;
190 }
191
192 // +1 to account for the mandatory transparency in Asperite color palettes
193 u32 pal_base = 1 + 2 * rarity;
194 subtype_info.main_color = card_rarity_pal_gfxPal[pal_base]; // even indices for shadows
195 subtype_info.shadow_color = card_rarity_pal_gfxPal[pal_base + 1]; // odd ones for main colors
196 subtype_info.name_str = JOKER_RARITY_STRINGS_LUT[rarity];
197
198 return subtype_info;
199}

216JokerObject* joker_object_new(Joker* joker)
217{
218 GBAL_RETURN_IF_NULL(joker, NULL);
219 JokerObject* joker_object = POOL_GET(JokerObject);
220
221 sprite_object_init((SpriteObject*)joker_object);
222
223 s16 layer = 0;
224 for (s16 i = 0; i < MAX_JOKER_OBJECTS; i++)
225 {
226 if (!s_used_layers[i])
227 {
228 layer = i;
229 s_used_layers[i] = true; // Mark this layer as used
230 break;
231 }
232 }
233
234 joker_object->joker = joker;
235 joker_object->type = ITEM_TYPE_JOKER;
236 joker_object->is_owned = false;
237
238 int tile_index = sprite_get_tid(JOKER_SPRITE, layer);
239 int joker_spritesheet_idx = s_joker_get_spritesheet_idx(joker->id);
240 int joker_idx = s_joker_get_sprite_idx_in_sheet(joker->id, joker_spritesheet_idx);
241 int joker_pb = s_allocate_pb_if_needed(joker->id);
242 s_joker_pb_add_sprite_user(joker_pb);
243
244 memcpy32(
245 &tile_mem[TILE_MEM_OBJ_CHARBLOCK0_IDX][tile_index],
246 &joker_gfxTiles[joker_spritesheet_idx][joker_idx * TILE_SIZE * JOKER_SPRITE_TILES],
247 TILE_SIZE * JOKER_SPRITE_TILES
248 );
249
250 sprite_object_set_sprite(
251 (SpriteObject*)joker_object,
252 sprite_new(
253 ATTR0_SQUARE | ATTR0_4BPP | ATTR0_AFF,
254 ATTR1_SIZE_32,
255 tile_index,
256 joker_pb,
257 sprite_get_starting_layer(JOKER_SPRITE) + layer
258 )
259 );
260
261 return joker_object;
262}

264void joker_object_destroy(JokerObject** joker_object)
265{
266 if (joker_object == NULL || *joker_object == NULL)
267 return;
268
269 s16 layer = sprite_get_layer(joker_object_get_sprite(*joker_object)) -
270 sprite_get_starting_layer(JOKER_SPRITE);
271 s_used_layers[layer] = false;
272 s_joker_pb_remove_sprite_user(sprite_get_pb(joker_object_get_sprite(*joker_object)));
273 if (s_joker_pb_get_num_sprite_users((sprite_get_pb(joker_object_get_sprite(*joker_object)))) ==
274 0)
275 {
276 s_joker_spritesheet_pb_map[s_joker_get_spritesheet_idx((*joker_object)->joker->id)] =
277 UNDEFINED;
278 }
279
280 sprite_object_destroy((SpriteObject*)(*joker_object));
281 joker_destroy(&(*joker_object)->joker);
282 POOL_FREE(JokerObject, *joker_object);
283 *joker_object = NULL;
284}

384int joker_roll_id(enum JokerRarity joker_rarity, enum RngSequence key)
385{
386 // Now determine how many jokers are available based on the rarity
387 int jokers_avail_size = get_num_rollable_jokers();
388
389 if (jokers_avail_size == 0)
390 return UNDEFINED;
391
392 int matching_joker_ids[jokers_avail_size];
393 int fallback_random_idx = rng_get_u32(key) % jokers_avail_size;
394 int fallback_random_joker_id = UNDEFINED;
395 int match_count = 0;
396
397 BitsetItr itr = bitset_itr_create(&s_rollable_jokers_bitset);
398
399 int i = 0;
400 int joker_id = UNDEFINED;
401 while ((joker_id = bitset_itr_next(&itr)) != UNDEFINED)
402 {
403 if (i++ == fallback_random_idx)
404 fallback_random_joker_id = joker_id;
405 const JokerInfo* info = get_joker_registry_entry(joker_id);
406 if (info->rarity == joker_rarity)
407 {
408 matching_joker_ids[match_count++] = joker_id;
409 }
410 }
411
412 int selected_joker_id = (match_count > 0) ? matching_joker_ids[rng_get_u32(key) % match_count]
413 : fallback_random_joker_id;
414
415 return selected_joker_id;
416}

418Item* joker_object_roll_new(enum RngSequence key)
419{
420 if (no_rollable_jokers())
421 return NULL;
422
423 int joker_id = 0;
424#ifdef TEST_JOKER_ID0 // Allow defining an ID for a joker to always appear in shop and be tested
425 if (joker_is_rollable(TEST_JOKER_ID0))
426 {
427 joker_id = TEST_JOKER_ID0;
428 }
429 else
430#endif

432 if (joker_is_rollable(TEST_JOKER_ID1))

434 joker_id = TEST_JOKER_ID1;

436 else
437#endif

457static inline enum JokerRarity joker_get_random_rarity(enum RngSequence key)
458{
459 enum JokerRarity joker_rarity = 0;
460 int rarity_roll = rng_get_u32(key) % 100;
461 if (rarity_roll < COMMON_JOKER_CHANCE)
462 {
463 joker_rarity = COMMON_JOKER;
464 }
465 else if (rarity_roll < COMMON_JOKER_CHANCE + UNCOMMON_JOKER_CHANCE)
466 {
467 joker_rarity = UNCOMMON_JOKER;
468 }
469 else if (rarity_roll < COMMON_JOKER_CHANCE + UNCOMMON_JOKER_CHANCE + RARE_JOKER_CHANCE)
470 {
471 joker_rarity = RARE_JOKER;
472 }
473 else if (rarity_roll < COMMON_JOKER_CHANCE + UNCOMMON_JOKER_CHANCE + RARE_JOKER_CHANCE +
474 LEGENDARY_JOKER_CHANCE)
475 {
476 joker_rarity = LEGENDARY_JOKER;
477 }
478
479 return joker_rarity;
480}
