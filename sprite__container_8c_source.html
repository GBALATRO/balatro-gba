27static inline void container_update(SpriteContainer* container)
28{
29 // position and size of the sprite along the main `direction`
30 POINT pos_size;
31 // Sprites' starting position along the main direction
32 int start_pos;
33 // size of the container along the main direction
34 int max_length;
35
36 if (container->orientation == LAYOUT_ORI_HORIZONTAL)
37 {
38 pos_size.x = container->sprite_local_visible_rect.left;
39 pos_size.y = rect_width(&(container->sprite_local_visible_rect)) - 1;
40
41 start_pos = container->pos.left;
42
43 max_length = rect_width(&(container->pos));
44 }
45 else
46 {
47 pos_size.x = container->sprite_local_visible_rect.top;
48 pos_size.y = rect_height(&(container->sprite_local_visible_rect)) - 1;
49
50 start_pos = container->pos.top;
51
52 max_length = rect_height(&(container->pos));
53 }
54
55 int nb_sprites = list_get_len(&container->contents_list);
56 int spacing = container->maximum_spacing;
57 int naive_length = nb_sprites * pos_size.y + (nb_sprites - 1) * spacing;
58
59 int overrun = naive_length - max_length;
60
61 container_adjust_pos_and_spacing(
62 container,
63 max_length,
64 overrun,
65 nb_sprites,
66 pos_size.y,
67 &spacing,
68 &start_pos
69 );
70
71 container_update_set_sprite_pos(container, pos_size, spacing, start_pos);
72
73 // Invoke custom callback if defined
74 if (container->custom_update)
75 container->custom_update(container);
76}

95static inline void container_adjust_pos_and_spacing(
96 SpriteContainer* container,
97 int max_length,
98 int overrun,
99 int nb_sprites,
100 int sprite_size,
101 int* spacing,
102 int* start_pos
103)
104{
105 // If sprites take too much space, correct the spacing
106 if (overrun > 0 && nb_sprites > 1)
107 {
108 // Ceil the reduction so the corrected layout does not still overflow.
109 int reduce = (overrun + (nb_sprites - 2)) / (nb_sprites - 1);
110 *spacing -= reduce;
111
112 // If the sprites need to be centered, and depending on the numer of them, the reduction to
113 // the spacing may cause an imbalance that can be solved by shifting the sprites slightly
114 // to the right
115 if (container->justification == LAYOUT_JUST_CENTER)
116 {
117 int new_length = nb_sprites * sprite_size + (nb_sprites - 1) * (*spacing);
118 *start_pos += (max_length - new_length) / 2;
119 }
120 }
121 // When they fit inside the container, correct the starting point if need be
122 else if (overrun < 0)
123 {
124 switch (container->justification)
125 {
126 case LAYOUT_JUST_CENTER:
127 *start_pos -= overrun / 2;
128 break;
129
130 case LAYOUT_JUST_END:
131 *start_pos -= overrun;
132 break;
133
134 default:
135 // Don't need to correct anything for the default alignment
136 break;
137 }
138 }
139 // And don't do anything if the fit is perfect (overrun == 0)
140}

152static inline void container_update_set_sprite_pos(
153 SpriteContainer* container,
154 POINT pos_size,
155 int spacing,
156 int start_pos
157)
158{
159 SpriteObject* sprite_object = NULL;
160 ListItr itr = list_itr_create(&container->contents_list);
161
162 while ((sprite_object = list_itr_next(&itr)))
163 {
164 FIXED* coord;
165 if (container->orientation == LAYOUT_ORI_HORIZONTAL)
166 coord = &sprite_object->tx;
167 else
168 coord = &sprite_object->ty;
169 *coord = int2fx(start_pos - pos_size.x);
170
171 start_pos += pos_size.y + spacing;
172 }
173}

224static inline int container_compute_ortho_sprite_pos(SpriteContainer* container)
225{
226 // These positions and dimentions are along the `direction` orthogonal to the main one
227 int size_ortho; // size of the sprite
228 int pos_ortho; // Sprites' starting position
229 int max_length_ortho; // size of the container
230
231 if (container->orientation == LAYOUT_ORI_HORIZONTAL)
232 {
233 size_ortho = rect_height(&(container->sprite_local_visible_rect)) - 1;
234 pos_ortho = container->pos.top;
235 max_length_ortho = rect_height(&(container->pos));
236 }
237 else
238 {
239 size_ortho = rect_width(&(container->sprite_local_visible_rect)) - 1;
240 pos_ortho = container->pos.left;
241 max_length_ortho = rect_width(&(container->pos));
242 }
243
244 int overrun_ortho = size_ortho - max_length_ortho;
245
246 // Always center along the direction orthogonal to the main one if Sprite takes too much
247 // space
248 if (overrun_ortho > 0)
249 pos_ortho -= overrun_ortho / 2;
250 // If there's enough room, align Sprite accoring to `container->justification_ortho`
251 else if (overrun_ortho < 0)
252 {
253 switch (container->justification_ortho)
254 {
255 case LAYOUT_JUST_CENTER:
256 pos_ortho -= overrun_ortho / 2;
257 break;
258
259 case LAYOUT_JUST_END:
260 pos_ortho -= overrun_ortho;
261 break;
262
263 default:
264 // Still don't need to correct anything for the default alignment
265 break;
266 }
267 }
268
269 return pos_ortho;
270}
