137#define TOP_UP_TAG_JOKER_BONUS 2
138static void skip_tag_effect_top_up(void)
139{
140 int available_joker_slots = MAX_JOKERS_HELD_SIZE - list_get_len(get_jokers_list());
141 if (available_joker_slots > TOP_UP_TAG_JOKER_BONUS)
142 available_joker_slots = TOP_UP_TAG_JOKER_BONUS;
143
144 for (int i = 0; i < available_joker_slots; i++)
145 {
146 int joker_id = joker_roll_id(COMMON_JOKER, RNG_SEQ_TAG_TOP_UP);
147
148 // Something went wrong, maybe there are no more Common Jokers available
149 if (joker_id == UNDEFINED)
150 break;
151
152 Joker* joker = joker_new(joker_id);
153 GBAL_RETURN_IF_NULL_VOID(joker);
154
155 joker_set_rollable(joker_id, false);
156 add_joker(joker_object_new(joker));
157 }
158}
