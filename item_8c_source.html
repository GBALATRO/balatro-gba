11Item* item_roll_new(enum ItemType item_type, enum RngSequence key)
12{
13 if ((int)item_type < 0 || item_type >= ITEM_NUM_TYPES)
14 {
15 MGBA_FUNC_ERROR("Invalid type %d", item_type);
16 return NULL;
17 }
18
19 ItemFuncs* item_funcs = get_item_type_funcs(item_type);
20
21 GBAL_RETURN_IF_NULL_RET(item_funcs, NULL);
22 if (item_funcs->roll_new == NULL)
23 {
24 MGBA_FUNC_ERROR("Unimplemented 'roll_new' function called for item type %d", item_type);
25 return NULL;
26 }
27
28 Item* ret_item = item_funcs->roll_new(key);
29 ret_item->is_owned = false;
30
31 return ret_item;
32}
