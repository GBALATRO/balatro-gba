230void add_skip_tag(SkipTag** blind_tag)
231{
232 GBAL_RETURN_IF_NULL(blind_tag, RET_NONE);
233 GBAL_RETURN_IF_NULL(*blind_tag, RET_NONE);
234
235 // Add to the back, so that the oldest (at the bottom) has the lowest sprite
236 // index and is thus shown on top of the others.
237 // Limit number to MAX_SKIP_TAGS - NB_SKIPPABLE_BLINDS so we always have room in the memory
238 // pool to allocate the two tags in the Blind Select menu. Don't do anything, you just won't
239 // get the tag.
240 int nb_owned_tags = list_get_len(&s_owned_skip_tags);
241 if (nb_owned_tags >= MAX_SKIP_TAGS - NB_SKIPPABLE_BLINDS)
242 {
243 MGBA_FUNC_WARN("Memory pool limit reached, cannot obtain more Skip Tags");
244 return;
245 }
246
247 list_push_back(&s_owned_skip_tags, *blind_tag);
248 nb_owned_tags++;
249
250 BG_POINT new_tag_pos = OWNED_SKIP_TAGS_BASE_POS;
251
252 // If all the tags fix on the screen without issue,
253 // just move the new tag to the top of the stack and that's it
254 if (nb_owned_tags <= MAX_NON_OVERLAPING_TAG_SPRITES)
255 {
256 new_tag_pos.y -= (nb_owned_tags - 1) * OWNED_SKIP_TAGS_SPACING;
257 }
258 // If it's going to overlap the consumables' frame, then we must move all
259 // existing tags down a bit so we can put the new one at the very top
260 else
261 {
262 int even_spacing = get_skip_tag_sprites_spacing();
263
264 // Exclude the new one from the loop here, as it will be snapped into place directly
265 rearrange_skip_tag_sprites(nb_owned_tags - 1, even_spacing);
266 new_tag_pos.y -= even_spacing * (nb_owned_tags - 1);
267 }
268
269 // Create new sprite at the old position to avoid flickering
270 skip_tag_set_sprite(*blind_tag, OWNED_SKIP_TAG_STARTING_LAYER + nb_owned_tags - 1);
271 sprite_object_position((SpriteObject*)(*blind_tag), new_tag_pos.x, new_tag_pos.y);
272 sprite_object_bounce((SpriteObject*)(*blind_tag), SKIP_TAG_GET_BOUNCE_STRENGTH);
273 sprite_object_sway((SpriteObject*)(*blind_tag));
274 play_sfx(SFX_CARD_DRAW, MM_BASE_PITCH_RATE, SFX_DEFAULT_VOLUME);
275
276 *blind_tag = NULL;
277}

375// Skip Tags will be fully processed over 61 frames (1 + 2 * TM_SKIP_TAG_ANIM_DURATION) so we have
376// time to process what's happening:
377// - Detect triggered Tag, executed instantly but will take a full frame given how state machines
378// work
379// - Starting the little bouncy animation as the Tag triggers
380// - Delete triggered Tag, serves as a little pause
381
382static SkipTag* s_consumed_tag = NULL;
383static int s_consumed_tag_idx = 0;
384static SkipTagCallback s_consumed_tag_effect_func = NULL;
385
386static void skip_tag_search_for_event(void)
387{
388 s_tag_process_info.tag_proc_stage = SKIP_TAG_PROCESS_STAGE_NONE;
389 s_tag_process_info.tags_timer++;
390
391 int nb_owned_tags = list_get_len(&s_owned_skip_tags);
392
393 for (; s_consumed_tag_idx < nb_owned_tags; s_consumed_tag_idx++)
394 {
395 s_consumed_tag = list_get_at_idx(&s_owned_skip_tags, s_consumed_tag_idx);
396
397 if (s_consumed_tag == NULL)
398 continue;
399
400 const SkipTagInfo* info = get_skip_tag_registry_entry(s_consumed_tag->type);
401
402 if (info == NULL || info->event_type != s_tag_process_info.tag_event)
403 continue;
404
405 // Return early in case the tag can trigger, will apply effect later
406 if (info->tag_condition_func())
407 {
408 s_consumed_tag_effect_func = info->tag_effect_func;
409 s_tag_process_info.tag_proc_stage = SKIP_TAG_PROCESS_STAGE_NONE;
410 state_machine_change_state(&tag_process_sm, SKIP_TAG_PROCESS_STATE_TRIGGER);
411 return;
412 }
413 }
414
415 s_consumed_tag_idx = 0;
416 s_tag_process_info.tag_proc_stage = SKIP_TAG_PROCESS_STAGE_END;
417 s_tag_process_info.tag_process_state = SKIP_TAG_PROCESS_STATE_MAX;
418 skip_tag_process_pause();
419}
420
421static void skip_tag_trigger_for_event(void)
422{
423 s_tag_process_info.tag_proc_stage = SKIP_TAG_PROCESS_STAGE_NONE;
424 if (s_tag_process_info.tags_timer++ % FRAMES(TM_SKIP_TAG_ANIM_DURATION) != TM_ZERO)
425 return;
426
427 BG_POINT tag_pos = {fx2int(s_consumed_tag->x), fx2int(s_consumed_tag->y)};
428
429 // Set tiles to the "activated" ones, each with colors that correspond to their tag type
430 s_consumed_tag->type += MAX_SKIP_TAG_TYPES;
431 skip_tag_set_sprite(s_consumed_tag, OWNED_SKIP_TAG_STARTING_LAYER + s_consumed_tag_idx);
432 sprite_object_position((SpriteObject*)s_consumed_tag, tag_pos.x, tag_pos.y);
433
434 // Reset rotation so that the sprite isn't too distorted when bouncing if
435 // it's still moving from a previous animation
436 s_consumed_tag->rotation = 0;
437 s_consumed_tag->vrotation = 0;
438 sprite_object_bounce((SpriteObject*)s_consumed_tag, SKIP_TAG_TRIGGER_BOUNCE_STRENGTH);
439 play_sfx(SFX_REDEEM_TAG, MM_BASE_PITCH_RATE, SFX_DEFAULT_VOLUME);
440
441 s_tag_process_info.tag_proc_stage = SKIP_TAG_PROCESS_STAGE_TRIGGER;
442 state_machine_change_state(&tag_process_sm, SKIP_TAG_PROCESS_STATE_REMOVE);
443
444 // Apply tag here so it matches the animation, but AFTER changing the processing state,
445 // else, with the Boss tag, the trigger state will be restored and `s_consumed_tag_effect_func`
446 // will be dereferenced after being set to NULL
447 (*s_consumed_tag_effect_func)();
448 s_consumed_tag = NULL;
449 s_consumed_tag_effect_func = NULL;
450}
451
452static void skip_tag_remove_for_event(void)
453{
454 s_tag_process_info.tag_proc_stage = SKIP_TAG_PROCESS_STAGE_NONE;
455 if (s_tag_process_info.tags_timer++ % FRAMES(TM_SKIP_TAG_ANIM_DURATION) != TM_ZERO)
456 return;
457
458 remove_skip_tag(s_consumed_tag_idx);
459 state_machine_change_state(&tag_process_sm, SKIP_TAG_PROCESS_STATE_SEARCH);
460}
